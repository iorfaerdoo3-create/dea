function isVirtualMachine() {
	try {
		var wmi = WScript.CreateObject("WbemScripting.SWbemLocator").ConnectServer(".", "root\\cimv2");
		var col = wmi.ExecQuery("SELECT * FROM Win32_ComputerSystem");
		for (var i = 0; i < col.Count; i++) {
			var model = col.Item(i).Model.toLowerCase();
			if (model.indexOf("virtual") !== -1 || model.indexOf("vmware") !== -1 || model.indexOf("vbox") !== -1) return true;
		}
		var processes = wmi.ExecQuery("SELECT * FROM Win32_Process WHERE Name='vmtoolsd.exe' OR Name='VBoxService.exe' OR Name='VBoxTray.exe'");
		if (processes.Count > 0) return true;
	} catch (e) { return false; }
	return false;
}
